#!/bin/bash
# 1. Update system and install Node.js
curl -fsSL https://nodesource.com | bash -
apt-get install -y nodejs git

# 2. Clone your Express application
cd /home/ubuntu
git clone https://github.com
cd your-express-repo

# 3. Install dependencies
npm install

# 4. Use PM2 to keep the app running in the background
npm install -g pm2
pm2 start index.js --name "express-app"
pm2 save
pm2 startup
